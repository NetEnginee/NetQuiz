        // --- Exponential Backoff Fetch Helper ---
        async function fetchWithBackoff(url, options, maxRetries = 3) {
            let delay = 1000;
            for (let i = 0; i < maxRetries; i++) {
                try {
                    const response = await fetch(url, options);
                    if (response.ok) return response;
                    if (response.status === 429) {
                        await new Promise(res => setTimeout(res, delay));
                        delay *= 2;
                        continue;
                    }
                    return response;
                } catch (err) {
                    if (i === maxRetries - 1) throw err;
                    await new Promise(res => setTimeout(res, delay));
                    delay *= 2;
                }
            }
        }

        // --- STREAMING_CHUNK:Implementing PCM to WAV audio converter for Gemini TTS API... ---
        // Helper: Convert PCM16 Int16Array to WAV Blob
        function pcmToWav(pcm16, sampleRate) {
            const buffer = new ArrayBuffer(44 + pcm16.length * 2);
            const view = new DataView(buffer);
            
            function writeString(v, offset, str) {
                for (let i = 0; i < str.length; i++) {
                    v.setUint8(offset + i, str.charCodeAt(i));
                }
            }

            writeString(view, 0, 'RIFF');
            view.setUint32(4, 36 + pcm16.length * 2, true);
            writeString(view, 8, 'WAVE');
            writeString(view, 12, 'fmt ');
            view.setUint32(16, 16, true);
            view.setUint16(20, 1, true); // PCM format
            view.setUint16(22, 1, true); // Mono
            view.setUint32(24, sampleRate, true);
            view.setUint32(28, sampleRate * 2, true);
            view.setUint16(32, 2, true);
            view.setUint16(34, 16, true);
            writeString(view, 36, 'data');
            view.setUint32(40, pcm16.length * 2, true);
            
            let offset = 44;
            for (let i = 0; i < pcm16.length; i++, offset += 2) {
                view.setInt16(offset, pcm16[i], true);
            }
            return new Blob([buffer], { type: 'audio/wav' });
        }

        function base64ToArrayBuffer(base64) {
            const binaryString = window.atob(base64);
            const len = binaryString.length;
            const bytes = new Uint8Array(len);
            for (let i = 0; i < len; i++) {
                bytes[i] = binaryString.charCodeAt(i);
            }
            return bytes.buffer;
        }

        async function playGeminiTtsText(textToSay, buttonIconId, buttonLabelId) {

                const payload = {

                        parts: [{ text: `Say in an upbeat, cheerful instructor voice: ${textToSay}` }]

                    generationConfig: {
                        responseModalities: ["AUDIO"],
                        speechConfig: {

                        }

                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(payload)

                const result = await res.json();
                const part = result?.candidates?.[0]?.content?.parts?.[0];
                const audioData = part?.inlineData?.data;
                const mimeType = part?.inlineData?.mimeType;

                if (audioData && mimeType) {
                    const sampleRate = parseInt(mimeType.match(/rate=(\d+)/)?.[1] || "24000", 10);
                    const arrayBuf = base64ToArrayBuffer(audioData);
                    const pcm16 = new Int16Array(arrayBuf);
                    const wavBlob = pcmToWav(pcm16, sampleRate);

                    const audio = new Audio(audioUrl);

                        if (iconEl) iconEl.textContent = '🔊';
                        if (labelEl) labelEl.textContent = 'Listen AI Voice';
                    };
                } else {
                    throw new Error("No audio returned");
                }
            } catch (err) {
                console.error("TTS Error:", err);
                if (iconEl) iconEl.textContent = '⚠️';
                if (labelEl) labelEl.textContent = 'Voice Unavailable';
                setTimeout(() => {
                    if (iconEl) iconEl.textContent = '🔊';

        function selectQuizAnswer(selectedIndex, btn) {
            const q = state.currentQuiz.questions[state.currentQuiz.currentIndex];
            const isCorrect = selectedIndex === q.correctIndex;

            // Disable all option buttons
            const allBtns = document.getElementById('quizOptionsContainer').querySelectorAll('button');
            allBtns.forEach((b, idx) => {
                b.disabled = true;
                if (idx === q.correctIndex) {
                    b.className = "w-full p-3 bg-emerald-950/80 border border-emerald-500 rounded-lg text-left font-mono text-xs text-emerald-200 flex justify-between";
                } else if (idx === selectedIndex) {
                    b.className = "w-full p-3 bg-red-950/80 border border-red-500 rounded-lg text-left font-mono text-xs text-red-200 flex justify-between";
                }
            });

            const expBox = document.getElementById('quizAnswerExplanation');
            expBox.classList.remove('hidden');

            if (isCorrect) {
                playPixelSound('coin');
                state.currentQuiz.score += 75;
                expBox.className = "p-3 rounded-lg border border-emerald-500/40 bg-emerald-950/30 text-emerald-300 font-mono text-xs mb-4";
                expBox.innerHTML = `<strong>✓ Correct! (+75 pts)</strong><br>${q.explanation}`;
            } else {
                playPixelSound('blip');
                expBox.className = "p-3 rounded-lg border border-red-500/40 bg-red-950/30 text-red-300 font-mono text-xs mb-4";
                expBox.innerHTML = `<strong>✗ Incorrect.</strong><br>${q.explanation}`;
            }

            document.getElementById('quizCurrentScore').textContent = `Score: ${state.currentQuiz.score} pts`;
            document.getElementById('nextQuestionBtn').classList.remove('hidden');
        }

        async function generateNetworkTopologyImage() {
            const prompt = document.getElementById('topologyPromptInput').value.trim();
            if (!prompt) return;

            document.getElementById('topologyPlaceholder').classList.add('hidden');
            document.getElementById('topologyImage').classList.add('hidden');
            document.getElementById('topologyLoading').classList.remove('hidden');
            playPixelSound('click');

            try {
                const apiKey = "";
                const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-3.1-flash-lite-image:generateContent?key=${apiKey}`;

                const payload = {
                    contents: [{ parts: [{ text: prompt }] }],
                    generationConfig: {
                        responseModalities: ['TEXT', 'IMAGE']
                    }
                };

                const res = await fetchWithBackoff(apiUrl, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(payload)
                });

                const result = await res.json();
                const base64Data = result?.candidates?.[0]?.content?.parts?.find(p => p.inlineData)?.inlineData?.data;

                if (base64Data) {
                    const imgEl = document.getElementById('topologyImage');
                    imgEl.src = `data:image/png;base64,${base64Data}`;
                    imgEl.classList.remove('hidden');
                    playPixelSound('badge');
                } else {
                    throw new Error("No image data returned");
                }
            } catch (err) {
                console.error("Image gen error:", err);
                document.getElementById('topologyPlaceholder').classList.remove('hidden');
                document.getElementById('topologyPlaceholder').innerHTML = `<p class="font-mono text-xs text-red-400">Error generating diagram image. Please try again.</p>`;
            } finally {
                document.getElementById('topologyLoading').classList.add('hidden');
            }
        }

        // --- Animated Canvas Network Data Stream Background ---
        const canvas = document.getElementById('networkCanvas');
        const ctx = canvas.getContext('2d');

        let width = canvas.width = window.innerWidth;
        let height = canvas.height = window.innerHeight;

        window.addEventListener('resize', () => {
            width = canvas.width = window.innerWidth;
            height = canvas.height = window.innerHeight;
        });

        class Packet {
            constructor() {
                this.reset();
            }

            reset() {
                const gridSize = 48;
                this.x = Math.floor(Math.random() * (width / gridSize)) * gridSize;
                this.y = Math.floor(Math.random() * (height / gridSize)) * gridSize;
                this.horizontal = Math.random() > 0.5;
                this.speed = (Math.random() * 1.5 + 1) * (Math.random() > 0.5 ? 1 : -1);
                this.size = 3;
                this.color = Math.random() > 0.5 ? '#0070f3' : '#50e3c2';
                this.life = 0;
                this.maxLife = Math.floor(Math.random() * 150 + 100);
            }

            update() {
                if (this.horizontal) {
                    this.x += this.speed;
                } else {
                    this.y += this.speed;
                }
                this.life++;
                if (this.life > this.maxLife || this.x < 0 || this.x > width || this.y < 0 || this.y > height) {
                    this.reset();
                }
            }

            draw() {
                ctx.fillStyle = this.color;
                for (let i = 0; i < 3; i++) {
                    const offset = i * 4;
                    const px = this.horizontal ? this.x - (offset * Math.sign(this.speed)) : this.x;
                    const py = this.horizontal ? this.y : this.y - (offset * Math.sign(this.speed));
                    ctx.globalAlpha = (1 - (i * 0.3)) * (1 - (this.life / this.maxLife));
                    ctx.fillRect(px, py, this.size, this.size);
                }
                ctx.globalAlpha = 1;
            }
        }

        const packets = Array.from({ length: 25 }, () => new Packet());

        function animateCanvas() {
            ctx.clearRect(0, 0, width, height);
            packets.forEach(p => {
                p.update();
                p.draw();
            });
            requestAnimationFrame(animateCanvas);
        }

        // Initialize Canvas Animation & UI State on window load
        window.onload = function() {
            animateCanvas();
            updateDashboardUI();
        };
    </script>
